      return document.querySelector('input[name="category"]:checked');

    function animateTotal(targetTotal) {
      const startTotal = currentDisplayedTotal;
      const diff = targetTotal - startTotal;
      if (diff === 0) {
        totalEl.innerHTML = arabicNumber.format(targetTotal) + ' <small>ر.س</small>';
        return;
      }
      const duration = 220;
      const startTime = performance.now();

      totalEl.style.transform = 'scale(1.06)';
      setTimeout(() => { totalEl.style.transform = 'scale(1)'; }, 160);

      function step(now) {
        const progress = Math.min((now - startTime) / duration, 1);
        const eased = 1 - Math.pow(1 - progress, 3);
        const current = Math.round(startTotal + diff * eased);
        totalEl.innerHTML = arabicNumber.format(current) + ' <small>ر.س</small>';
        if (progress < 1) {
          requestAnimationFrame(step);
        } else {
          currentDisplayedTotal = targetTotal;
        }
      }
      requestAnimationFrame(step);
    }

    function updateSummary() {
      const ticket = selectedTicket();
      if (!ticket) return;
      document.querySelectorAll('.package-card').forEach(card => {
        card.classList.toggle('selected', card.contains(ticket));

      const price = Number(ticket.dataset.price);
      const total = price * quantity;

      quantityEl.textContent = arabicNumber.format(quantity);
      document.getElementById('summary-category').textContent = ticket.value;
      document.getElementById('summary-unit').textContent = arabicNumber.format(price) + ' ر.س';
      document.getElementById('summary-quantity').textContent = arabicNumber.format(quantity);
      animateTotal(total);
    }

      statusEl.textContent = message;

      const orbBtn = document.getElementById('music-orb-btn');
      const orbLabel = document.getElementById('music-orb-label');
      const timeBadge = document.getElementById('music-time-badge');
      const audioEl = document.getElementById('evening-audio');
      const filePicker = document.getElementById('audio-file-picker');
      const ringProgress = document.getElementById('orb-ring-progress');
      const ringThumb = document.getElementById('orb-ring-thumb');
      const ringHit = document.getElementById('orb-ring-hit');

      const radius = 42;
      const center = 46;
      const circumference = 2 * Math.PI * radius; // 263.89

      let isScrubbing = false;
      let justScrubbed = false;
      let scrubRatio = 0;

      function formatTime(sec) {
        if (!Number.isFinite(sec) || sec < 0) return '0:00';
        const m = Math.floor(sec / 60);
        const s = Math.floor(sec % 60);
        return `${m}:${s < 10 ? '0' : ''}${s}`;
      }

      function setCircleProgress(ratio) {
        const clamped = Math.max(0, Math.min(1, ratio));
        const offset = circumference - clamped * circumference;
        ringProgress.style.strokeDashoffset = offset.toFixed(2);

        const angleRad = clamped * 2 * Math.PI;
        const tx = center + radius * Math.cos(angleRad);
        const ty = center + radius * Math.sin(angleRad);
        ringThumb.setAttribute('cx', tx.toFixed(2));
        ringThumb.setAttribute('cy', ty.toFixed(2));
      }

      function getRatioFromPointer(clientX, clientY) {
        const rect = orbBtn.getBoundingClientRect();
        const cx = rect.left + rect.width / 2;
        const cy = rect.top + rect.height / 2;
        const dx = clientX - cx;
        const dy = clientY - cy;

        let angle = Math.atan2(dx, -dy);
        if (angle < 0) angle += 2 * Math.PI;
        return angle / (2 * Math.PI);
      }

      function applyScrub(clientX, clientY, commitAudio) {
        scrubRatio = getRatioFromPointer(clientX, clientY);
        setCircleProgress(scrubRatio);
        orbWrapper.classList.add('has-progress');

        const duration = audioEl.duration;
        if (Number.isFinite(duration) && duration > 0) {
          const targetTime = scrubRatio * duration;
          timeBadge.textContent = `${formatTime(targetTime)} / ${formatTime(duration)}`;
          if (commitAudio) {
            audioEl.currentTime = targetTime;
          }
        }
      }

      function setPlayingUI(playing) {
        isMusicPlaying = playing;
        orbWrapper.classList.toggle('is-playing', playing);
        orbBtn.setAttribute('aria-pressed', String(playing));
        orbLabel.textContent = playing
          ? 'أغنيـة البُوشيــة'
          : 'إستئناف أغنيـة البُوشيـة';
      }

      ringHit.addEventListener('pointerdown', e => {
        e.stopPropagation();
        isScrubbing = true;
        justScrubbed = true;
        orbWrapper.classList.add('is-scrubbing');
        if (ringHit.setPointerCapture) {
          try { ringHit.setPointerCapture(e.pointerId); } catch (_) {}
        }
        applyScrub(e.clientX, e.clientY, true);
      });

      window.addEventListener('pointermove', e => {
        if (!isScrubbing) return;
        applyScrub(e.clientX, e.clientY, true);
      }, { passive: true });

      function endScrub(e) {
        if (!isScrubbing) return;
        if (e && typeof e.clientX === 'number') {
          applyScrub(e.clientX, e.clientY, true);
        }
        isScrubbing = false;
        orbWrapper.classList.remove('is-scrubbing');
        setTimeout(() => { justScrubbed = false; }, 120);
      }

      window.addEventListener('pointerup', endScrub, { passive: true });
      window.addEventListener('pointercancel', endScrub, { passive: true });

      orbBtn.addEventListener('click', async e => {
        if (justScrubbed || isScrubbing) {
          e.preventDefault();
          return;
        }

        if (!isMusicPlaying) {
          try {
            await audioEl.play();
            setPlayingUI(true);
          } catch (err) {
            filePicker.click();
          }
        } else {
          audioEl.pause();
          setPlayingUI(false);
        }
      });

      filePicker.addEventListener('change', async e => {
        const file = e.target.files && e.target.files[0];
        if (!file) return;
        const objectUrl = URL.createObjectURL(file);
        audioEl.src = objectUrl;
        try {
          await audioEl.play();
          setPlayingUI(true);
        } catch (playErr) {
          setPlayingUI(false);
        }
      });

      audioEl.addEventListener('loadedmetadata', () => {
        if (audioEl.duration) {
          timeBadge.textContent = `0:00 / ${formatTime(audioEl.duration)}`;
        }
      });

      audioEl.addEventListener('timeupdate', () => {
        if (!isScrubbing && audioEl.duration) {
          const ratio = audioEl.currentTime / audioEl.duration;
          setCircleProgress(ratio);
          if (audioEl.currentTime > 0) orbWrapper.classList.add('has-progress');
          timeBadge.textContent = `${formatTime(audioEl.currentTime)} / ${formatTime(audioEl.duration)}`;
        }
      });

      audioEl.addEventListener('pause', () => {
        if (isMusicPlaying) setPlayingUI(false);
      });

      audioEl.addEventListener('play', () => {
        if (!isMusicPlaying) setPlayingUI(true);
      });
    })();

    /* ==========================================================================
       تفاعل المربعات بالضغط واللمس والتحويم (Interactive Pop & Spotlight Cards)
       ========================================================================== */
    document.querySelectorAll('.spotlight-card').forEach(card => {
      function updateCardSpot(clientX, clientY) {
        const rect = card.getBoundingClientRect();
        card.style.setProperty('--mouse-x', `${(clientX - rect.left).toFixed(1)}px`);
        card.style.setProperty('--mouse-y', `${(clientY - rect.top).toFixed(1)}px`);
      }

      card.addEventListener('pointermove', e => updateCardSpot(e.clientX, e.clientY), { passive: true });

      card.addEventListener('pointerdown', e => {
        updateCardSpot(e.clientX, e.clientY);
        card.classList.add('is-pressed');
      }, { passive: true });

      card.addEventListener('pointerup', () => {
        setTimeout(() => card.classList.remove('is-pressed'), 180);
      }, { passive: true });

      card.addEventListener('pointerleave', () => {
        card.classList.remove('is-pressed');
      }, { passive: true });
    });

    /* ==========================================================================
       انبثاق المربعات عند التمرير للأعلى أو الأسفل (Scroll Pop Observer)
       ========================================================================== */
    const observer = new IntersectionObserver(entries => {
      entries.forEach(entry => {
        if (entry.isIntersecting) {
          entry.target.classList.add('is-visible');
        } else {
          entry.target.classList.remove('is-visible');
        }
      });
    }, { threshold: 0.12 });

    document.querySelectorAll('.reveal').forEach(el => observer.observe(el));

    /* ==========================================================================
       محرك الحركة الدائمة: الصورة الأصلية المتحركة + خلفية القماش الحريري التفاعلي
       ========================================================================== */
    (function initContinuousSilkAndHeroEngine() {
      const canvas = document.getElementById('silk-canvas');
      const ctx = canvas ? canvas.getContext('2d') : null;
      let width = window.innerWidth;
      let height = window.innerHeight;

      let pointerX = width * 0.5;
      let pointerY = height * 0.4;
      let smoothX = pointerX;
      let smoothY = pointerY;

      let heroPointerX = 0;
      let heroPointerY = 0;
      let smoothHeroX = 0;
      let smoothHeroY = 0;

      function resize() {
        width = window.innerWidth;
        height = window.innerHeight;
        if (canvas) {
          canvas.width = width;
          canvas.height = height;
        }
      }
      window.addEventListener('resize', resize, { passive: true });
      resize();

      function onMove(x, y) {
        pointerX = x;
        pointerY = y;
        if (heroSection) {
          const rect = heroSection.getBoundingClientRect();
          if (rect.bottom > 0) {
            const rx = (x - rect.left) / rect.width - 0.5;
            const ry = (y - rect.top) / rect.height - 0.5;
            heroPointerX = rx * -22;
            heroPointerY = ry * -14;
          }
        }
      }

      window.addEventListener('pointermove', e => onMove(e.clientX, e.clientY), { passive: true });
      window.addEventListener('touchmove', e => {
        if (e.touches[0]) onMove(e.touches[0].clientX, e.touches[0].clientY);
      }, { passive: true });

      window.addEventListener('scroll', () => {
        const docHeight = document.documentElement.scrollHeight - window.innerHeight;
        progressEl.style.width = docHeight > 0 ? `${(window.scrollY / docHeight) * 100}%` : '0%';
      }, { passive: true });

      const silkRibbons = [
        { yRatio: 0.22, amp: 38, freq: 0.0022, speed: 0.014, alpha: 0.12, thickness: 150 },
        { yRatio: 0.42, amp: 48, freq: 0.0018, speed: -0.011, alpha: 0.10, thickness: 180 },
        { yRatio: 0.62, amp: 42, freq: 0.0025, speed: 0.016, alpha: 0.11, thickness: 160 },
        { yRatio: 0.80, amp: 35, freq: 0.0020, speed: -0.013, alpha: 0.09, thickness: 140 },
        { yRatio: 0.92, amp: 28, freq: 0.0028, speed: 0.018, alpha: 0.08, thickness: 120 }
      ];

      let time = 0;

      function renderLoop() {
        time += isMusicPlaying ? 1.5 : 1;

        smoothX += (pointerX - smoothX) * 0.06;
        smoothY += (pointerY - smoothY) * 0.06;
        smoothHeroX += (heroPointerX - smoothHeroX) * 0.05;
        smoothHeroY += (heroPointerY - smoothHeroY) * 0.05;

        if (heroSection) {
          const autoX = Math.sin(time * 0.015) * 11 + Math.cos(time * 0.008) * 5;
          const autoY = Math.cos(time * 0.012) * 7;
          const autoScale = 1.06 + Math.sin(time * 0.01) * 0.018;

          const totalHeroX = autoX + smoothHeroX;
          const totalHeroY = autoY + smoothHeroY;

          heroSection.style.setProperty('--hero-x', `${totalHeroX.toFixed(2)}px`);
          heroSection.style.setProperty('--hero-y', `${totalHeroY.toFixed(2)}px`);
          heroSection.style.setProperty('--hero-scale', autoScale.toFixed(4));

          const spotX = 32 + Math.sin(time * 0.014) * 14 + (smoothHeroX * -0.8);
          const spotY = 45 + Math.cos(time * 0.011) * 12 + (smoothHeroY * -0.8);
          heroSection.style.setProperty('--hero-spot-x', `${spotX.toFixed(1)}%`);
          heroSection.style.setProperty('--hero-spot-y', `${spotY.toFixed(1)}%`);
        }

        if (ctx) {
          ctx.clearRect(0, 0, width, height);

          const pointerGlow = ctx.createRadialGradient(smoothX, smoothY, 10, smoothX, smoothY, 380);
          pointerGlow.addColorStop(0, 'rgba(245, 187, 103, 0.085)');
          pointerGlow.addColorStop(0.5, 'rgba(201, 128, 46, 0.03)');
          pointerGlow.addColorStop(1, 'transparent');
          ctx.fillStyle = pointerGlow;
          ctx.fillRect(0, 0, width, height);

          const scrollOffset = window.scrollY * 0.15;

          for (let r = 0; r < silkRibbons.length; r++) {
            const rib = silkRibbons[r];
            const baseY = (height * rib.yRatio) - (scrollOffset % (height * 0.35));
            const phase = time * rib.speed + r * 1.7;

            const grad = ctx.createLinearGradient(0, baseY - rib.thickness, width, baseY + rib.thickness);
            grad.addColorStop(0, 'rgba(201, 128, 46, 0)');
            grad.addColorStop(0.25, `rgba(245, 187, 103, ${rib.alpha * 0.65})`);
            grad.addColorStop(0.5, `rgba(255, 244, 217, ${rib.alpha})`);
            grad.addColorStop(0.75, `rgba(201, 128, 46, ${rib.alpha * 0.65})`);
            grad.addColorStop(1, 'rgba(201, 128, 46, 0)');

            ctx.beginPath();
            const stepX = 28;

            for (let x = 0; x <= width + stepX; x += stepX) {
              const distToPointer = Math.hypot(x - smoothX, baseY - smoothY);
              const pointerPull = Math.max(0, 1 - distToPointer / 360) * ((smoothY - baseY) * 0.28);

              const waveY =
                baseY +
                Math.sin(x * rib.freq + phase) * rib.amp +
                Math.cos(x * rib.freq * 0.6 - phase * 1.3) * (rib.amp * 0.45) +
                pointerPull;

              if (x === 0) ctx.moveTo(x, waveY);
              else ctx.lineTo(x, waveY);
            }

            for (let x = width + stepX; x >= 0; x -= stepX) {
              const distToPointer = Math.hypot(x - smoothX, baseY - smoothY);
              const pointerPull = Math.max(0, 1 - distToPointer / 360) * ((smoothY - baseY) * 0.28);

              const lowerWaveY =
                baseY +
                rib.thickness * (0.55 + 0.35 * Math.sin(x * 0.003 + phase)) +
                Math.sin(x * rib.freq + phase + 0.9) * (rib.amp * 0.85) +
                pointerPull;

              ctx.lineTo(x, lowerWaveY);
            }

            ctx.closePath();
            ctx.fillStyle = grad;
            ctx.fill();

            ctx.beginPath();
            for (let x = 0; x <= width + stepX; x += stepX) {
              const distToPointer = Math.hypot(x - smoothX, baseY - smoothY);
              const pointerPull = Math.max(0, 1 - distToPointer / 360) * ((smoothY - baseY) * 0.28);
              const waveY =
                baseY +
                Math.sin(x * rib.freq + phase) * rib.amp +
                Math.cos(x * rib.freq * 0.6 - phase * 1.3) * (rib.amp * 0.45) +
                pointerPull;
              if (x === 0) ctx.moveTo(x, waveY);
              else ctx.lineTo(x, waveY);
            }
            ctx.strokeStyle = `rgba(255, 240, 205, ${rib.alpha * 1.35})`;
            ctx.lineWidth = 1;
            ctx.stroke();
          }
        }

        requestAnimationFrame(renderLoop);
      }

      requestAnimationFrame(renderLoop);
    })();

    async function initialize() {
      try {
        if (window.dataSdk && typeof window.dataSdk.init === 'function') {
          const result = await window.dataSdk.init({ onDataChanged() {} });
          sdkReady = Boolean(result && result.isOk);
        }
      } catch (e) {
        sdkReady = false;
      }
    }

    form.addEventListener('submit', async event => {
      event.preventDefault();

      const nameInput = document.getElementById('full-name');
      const emailInput = document.getElementById('email');
      const phoneInput = document.getElementById('phone');
      let hasError = false;

      [nameInput, emailInput, phoneInput].forEach(input => {
        if (!input.value.trim() || !input.checkValidity()) {
          input.classList.add('input-error');
          hasError = true;
        }
      });

      if (hasError) {
        showStatus('يرجى التحقق من إكمال جميع بيانات التواصل بشكل صحيح قبل إرسال الطلب.', 'error');
        return;
      }

      const ticket = selectedTicket();
      const record = {
        full_name: nameInput.value.trim(),
        email: emailInput.value.trim(),
        phone: phoneInput.value.trim(),
        ticket_category: ticket.value,
        quantity,
        estimated_total: Number(ticket.dataset.price) * quantity,
        submitted_at: new Date().toISOString()
      };

      submitButton.disabled = true;
      submitButton.innerHTML = '<span>جارٍ إرسال الطلب...</span>';
      statusEl.className = 'status';

      try {
        if (sdkReady && window.dataSdk) {
          const result = await window.dataSdk.create(record);
          if (result && result.isOk) {
            showStatus('تم إرسال طلب حجزكِ بنجاح! سيتواصل معكِ فريق التنظيم لتأكيد التوفر والتفاصيل.', 'success');
            form.reset();
            quantity = 1;
            updateSummary();
          } else {
            showStatus('لم يُرسل الطلب. يرجى المحاولة مرة أخرى بعد قليل.', 'error');
          }
        } else {
          await new Promise(resolve => setTimeout(resolve, 480));
          showStatus('تم استلام طلب حجزكِ بنجاح! سيتواصل معكِ فريق التنظيم قريبًا لتأكيد التوفر والتفاصيل.', 'success');
          form.reset();
          quantity = 1;
          updateSummary();
        }
      } catch (e) {
        showStatus('حدث خطأ أثناء إرسال الطلب. يرجى المحاولة مرة أخرى.', 'error');
      } finally {
        submitButton.disabled = false;
        submitButton.innerHTML = '<span>إرسال طلب الحجز</span><svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2.4" stroke-linecap="round" stroke-linejoin="round" aria-hidden="true"><path d="m12 19-7-7 7-7"/><path d="M19 12H5"/></svg>';
      }
    });

    updateSummary();
    initialize();
  </script>
